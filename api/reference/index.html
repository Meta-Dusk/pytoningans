    class BaseEntityBehavior:
    """
    Inherit from this class in your entity's `behavior.py`.
    Override any of the hook methods below to implement custom AI logic, interactions,
    combat rules, or dialogue responses.
    """
    def on_spawn(self, entity: IEntity) -> None:
        """Called once when the entity is instantiated and placed in a world."""
        pass

    def on_update(self, entity: IEntity, dt: int) -> None:
        """Called every frame tick. Use for real-time timers and state monitoring."""
        pass

    def on_decision_tick(self, entity: IEntity) -> bool:
        """
        Called when the engine's AI brain evaluates a wander or action decision.
        Return True to override default engine movement/action behavior.
        """
        return False

    def on_interact(self, entity: IEntity, other_entity: IEntity) -> bool:
        """
        Called when this entity collides with or approaches another living entity.
        Return True to override the default social interaction.
        """
        return False

    def on_item_interact(self, entity: IEntity, item: IItem) -> bool:
        """
        Called when this entity collides with or touches an item on the ground.
        Return True to override default collision handling (e.g. automatically picking up).
        """
        return False

    def on_seek_item(self, entity: IEntity, available_items: list[IItem]) -> Optional[IItem] | bool:
        """
        Called when the entity evaluates seeking an item (e.g. survival or scavenging).
        - Return a specific IItem to actively pathfind towards it.
        - Return True to indicate custom seeking was handled (skips default search).
        - Return False or None to allow default survival seeking (e.g. seeking healing items when low on HP) to run.
        """
        return None

    def on_speak(self, entity: IEntity, text: str) -> None:
        """Called whenever this entity speaks dialogue via say() or force_talk()."""
        pass

    def on_damage(self, entity: IEntity, amount: int) -> None:
        """Called when the entity takes damage."""
        pass

    def on_death(self, entity: IEntity) -> None:
        """Called immediately after the entity's health drops to 0."""
        pass

    def on_revive(self, entity: IEntity) -> None:
        """Called when the entity is revived from death."""
        pass

    def on_attack(self, entity: IEntity, target: IEntity) -> bool:
        """
        Called when the entity initiates an attack against target.
        Return True to override engine combat resolution.
        """
        return False

    def on_state_change(self, entity: IEntity, old_state: str, new_state: str) -> None:
        """Called whenever the entity's state changes (e.g. idle -> moving)."""
        pass

    def on_hover_enter(self, entity: IEntity) -> None:
        """Called when the mouse cursor enters the entity's bounding area."""
        pass

    def on_hover_leave(self, entity: IEntity) -> None:
        """Called when the mouse cursor leaves the entity's bounding area."""
        pass

    def on_cursor_grabbed(self, entity: IEntity) -> None:
        """Called when this entity grabs control of the OS mouse cursor."""
        pass

    def on_cursor_released(self, entity: IEntity) -> None:
        """Called when this entity releases the OS mouse cursor."""
        pass

    class SettingsManager(QObject):
    settings_changed = Signal()
    _instance: Optional[SettingsManager] = None

    def __init__(self) -> None:
        super().__init__()
        self.settings: AppSettings = AppSettings()
        self.load()

    @classmethod
    def get(cls) -> SettingsManager:
        if cls._instance is None:
            cls._instance = SettingsManager()
        return cls._instance

    def load(self) -> None:
        path: Path = get_settings_path()
        if path.exists():
            try:
                data: dict[str, Any] = json.loads(path.read_text(encoding="utf-8"))
                self.settings = AppSettings(
                    health_bar_mode=data.get("health_bar_mode", HealthBarMode.ON_DAMAGE),
                    show_damage_numbers=data.get("show_damage_numbers", True),
                    show_health_numbers=data.get("show_health_numbers", True),
                    enable_health_shake=data.get("enable_health_shake", True),
                    auto_hide_delay_ms=data.get("auto_hide_delay_ms", 3000),
                    enable_window_physics=data.get("enable_window_physics", True),
                    avoid_window_spawn=data.get("avoid_window_spawn", True),
                    spawn_area_mode=data.get("spawn_area_mode", "screen"),
                    enable_debug_mode=data.get("enable_debug_mode", False),
                    creator_name=data.get("creator_name", "Modder"),
                    creator_bio=data.get("creator_bio", ""),
                    creator_avatar_path=data.get("creator_avatar_path", ""),
                )
            except Exception as e:
                print(f"Warning: Failed to load settings from {path}: {e}")
                self.settings = AppSettings()
        else:
            self.settings = AppSettings()

    def save(self) -> None:
        path: Path = get_settings_path()
        try:
            path.write_text(json.dumps(asdict(self.settings), indent=2), encoding="utf-8")
        except Exception as e:
            print(f"Warning: Failed to save settings to {path}: {e}")

    def update_settings(
        self,
        health_bar_mode: Optional[str] = None,
        show_damage_numbers: Optional[bool] = None,
        show_health_numbers: Optional[bool] = None,
        enable_health_shake: Optional[bool] = None,
        auto_hide_delay_ms: Optional[int] = None,
        enable_window_physics: Optional[bool] = None,
        avoid_window_spawn: Optional[bool] = None,
        spawn_area_mode: Optional[str] = None,
        enable_debug_mode: Optional[bool] = None,
        creator_name: Optional[str] = None,
        creator_bio: Optional[str] = None,
        creator_avatar_path: Optional[str] = None,
    ) -> None:
        changed: bool = False
        if health_bar_mode is not None and self.settings.health_bar_mode != health_bar_mode:
            self.settings.health_bar_mode = health_bar_mode
            changed = True
        if show_damage_numbers is not None and self.settings.show_damage_numbers != show_damage_numbers:
            self.settings.show_damage_numbers = show_damage_numbers
            changed = True
        if show_health_numbers is not None and self.settings.show_health_numbers != show_health_numbers:
            self.settings.show_health_numbers = show_health_numbers
            changed = True
        if enable_health_shake is not None and self.settings.enable_health_shake != enable_health_shake:
            self.settings.enable_health_shake = enable_health_shake
            changed = True
        if auto_hide_delay_ms is not None and self.settings.auto_hide_delay_ms != auto_hide_delay_ms:
            self.settings.auto_hide_delay_ms = auto_hide_delay_ms
            changed = True
        if enable_window_physics is not None and self.settings.enable_window_physics != enable_window_physics:
            self.settings.enable_window_physics = enable_window_physics
            changed = True
        if avoid_window_spawn is not None and self.settings.avoid_window_spawn != avoid_window_spawn:
            self.settings.avoid_window_spawn = avoid_window_spawn
            changed = True
        if spawn_area_mode is not None and self.settings.spawn_area_mode != spawn_area_mode:
            self.settings.spawn_area_mode = spawn_area_mode
            changed = True
        if enable_debug_mode is not None and self.settings.enable_debug_mode != enable_debug_mode:
            self.settings.enable_debug_mode = enable_debug_mode
            changed = True
        if creator_name is not None and self.settings.creator_name != creator_name.strip():
            self.settings.creator_name = creator_name.strip()
            changed = True
        if creator_bio is not None and self.settings.creator_bio != creator_bio.strip():
            self.settings.creator_bio = creator_bio.strip()
            changed = True
        if creator_avatar_path is not None and self.settings.creator_avatar_path != creator_avatar_path.strip():
            self.settings.creator_avatar_path = creator_avatar_path.strip()
            changed = True

        if changed:
            self.save()
            self.settings_changed.emit()

    def reset_to_defaults(self) -> None:
        self.settings = AppSettings()
        self.save()
        self.settings_changed.emit()
