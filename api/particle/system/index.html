    @runtime_checkable
class IParticleManager(Protocol):
    """Public interface for spawning particles, and creating emitters."""

    active_particles: list[IParticle]
    active_emitters: list[IParticleEmitter]

    def spawn(
        self,
        pixmap: ResolvableTexture,
        pos: Pos2D,
        velocity: Pos2D,
        config: Optional[ParticleConfigDTO] = None,
        parent: Optional[IGraphicsItem] = None,
    ) -> IParticle:
        ...

    def spawn_projectile(
        self,
        pixmap: ResolvableTexture,
        pos: Pos2D,
        velocity: Pos2D,
        config: Optional[ParticleConfigDTO] = None,
        parent: Optional[IGraphicsItem] = None,
    ) -> IParticle:
        """Spawns an interactive projectile pre-configured with collision and 'projectile' tag."""
        ...

    def spawn_animated_particle(
        self,
        sprite_sheet: ResolvableTexture,
        pos: Pos2D,
        velocity: Pos2D,
        frame_width: int,
        frame_height: int,
        frame_count: int,
        fps: float,
        loop: bool = True,
        config: Optional[ParticleConfigDTO] = None,
        parent: Optional[IGraphicsItem] = None,
    ) -> IAnimatedParticle:
        ...

    def create_emitter(self, config: ParticleEmitterConfigDTO) -> IParticleEmitter:
        """Creates a continuous stream or timed duration particle emitter."""
        ...

    def clear_all(self) -> None: ...

    @runtime_checkable
class IParticleEmitter(Protocol):
    """Interface representing a continuous stream, burst, or trail particle emitter."""
    config: ParticleEmitterConfigDTO
    pos: Pos2D
    target: Optional[IGraphicsItem]
    time_active: float
    emission_accumulator: float
    is_finished: bool
    is_active: bool
    sub_emitters: list[IParticleEmitter]

    def burst(self, count: int) -> list[IParticle]:
        """Instantly emits a batch of particles with randomized properties."""
        ...

    def destroy(self) -> None:
        """Stops the emitter and marks it for scene removal."""
        ...
